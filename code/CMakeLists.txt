cmake_minimum_required(VERSION 3.16)
project(Lab5Variant7 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(task RANGE 1 5)

    add_executable(test_5_${task} test_5_${task}.cpp)
    add_test(NAME task_5_${task} COMMAND test_5_${task})
endforeach()
